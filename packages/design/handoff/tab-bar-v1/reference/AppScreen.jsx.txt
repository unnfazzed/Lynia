import React from "react";
import { StatusBar } from "./StatusBar.jsx";
import { TabBar, TAB_BAR_SPACE, TAB_BAR_GAP } from "./TabBar.jsx";

/**
 * Phone screen scaffold — status bar, optional banner strip, body, sticky footer, root tab bar.
 * Every app screen in the mocks and the prototype sits in one of these, so chrome never drifts
 * between journeys. With a tab bar, a footer becomes the "dock": one white panel whose bottom
 * padding holds the floating bar (CTA → 12 → bar → 12 + inset).
 */
export function AppScreen({ children, footer, tab, tabs, tabDot, tabBadges, role, inset = 0, banner, bg = "var(--bg)", dark = false, onTab, style, ...rest }) {
  const space = TAB_BAR_SPACE + inset;
  return (
    <div style={{ position: "relative", height: "100%", background: bg, display: "flex", flexDirection: "column", overflow: "hidden", fontFamily: "var(--font-sans)", ...style }} {...rest}>
      <StatusBar dark={dark} />
      {banner}
      <div style={{ flex: 1, minHeight: 0, overflowY: "auto", overflowX: "hidden", paddingBottom: tab && !footer ? space + 16 : 0, scrollbarWidth: "none", msOverflowStyle: "none" }}>{children}</div>
      {footer ? <div style={tab ? { padding: `12px 16px ${TAB_BAR_GAP + space}px`, background: "var(--bg)", boxShadow: "var(--shadow-sheet)", position: "relative", zIndex: 1 } : { padding: "8px 16px 12px", background: "var(--bg)", borderTop: "1px solid var(--line)" }}>{footer}</div> : null}
      {tab ? <TabBar active={tab} role={role} tabs={tabs} dot={tabDot} badges={tabBadges} inset={inset} onTab={onTab} /> : null}
    </div>
  );
}
